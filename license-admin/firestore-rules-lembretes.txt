// junto com as regras de "users" e outras coleções.

match /lembretes/{lembreteId} {
  allow read, delete: if request.auth != null && request.auth.uid == resource.data.uid;
  allow create: if request.auth != null && request.auth.uid == request.resource.data.uid;
  // updates só pelo backend (Cloud Function usa Admin SDK, que ignora as rules)
  allow update: if false;
}

match /users/{userId} {
  allow read, write: if request.auth != null && request.auth.uid == userId;
}
